# Vitox M2 - out-of-tree WindowsStore build of Dynarmic (A32 frontend, x64 backend).
# Reads C:\Vitox\Vita3K\external\dynarmic as-is; nothing is written into the Vita3K tree.
cmake_minimum_required(VERSION 3.20)
project(vitox_m2_dynarmic CXX C)

set(VITOX_DYNARMIC_SOURCE "C:/Vitox/Vita3K/external/dynarmic" CACHE PATH "Dynarmic source (read-only)")
set(VITOX_BOOST_INCLUDE "C:/Vitox/Vita3K/external/boost" CACHE PATH "Boost headers (header-only use)")

if (NOT CMAKE_SYSTEM_NAME STREQUAL "WindowsStore")
    message(FATAL_ERROR "Configure with -DCMAKE_SYSTEM_NAME=WindowsStore -DCMAKE_SYSTEM_VERSION=10.0")
endif()

# Same choices as Vita3K's external/CMakeLists.txt, plus W^X support for the AppContainer.
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_FRONTENDS "A32" CACHE STRING "" FORCE)
set(DYNARMIC_ENABLE_NO_EXECUTE_SUPPORT ON CACHE BOOL "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_USE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)

# Boost is header-only for Dynarmic.
set(Boost_INCLUDE_DIR "${VITOX_BOOST_INCLUDE}" CACHE PATH "" FORCE)
set(Boost_NO_SYSTEM_PATHS ON)

add_subdirectory("${VITOX_DYNARMIC_SOURCE}" "${CMAKE_BINARY_DIR}/dynarmic")

# ---- Vitox shim: W^X ordering fix in A32EmitX64's constructor -------------------------
# Upstream calls code.PreludeComplete() (which flips the code cache to RX under
# DYNARMIC_ENABLE_NO_EXECUTE_SUPPORT) *before* exception_handler.SetFastmemCallback(),
# and on Windows SetFastmemCallback writes UNW_EXCEPTION_INFO::ExceptionHandler, which
# lives inside the code cache. Result: write access violation in every A32::Jit
# constructor. The fix registers the callback while the cache is still writable.
# The Vita3K tree is not touched: a patched copy is compiled in place of the original.
set(_vitox_src "${VITOX_DYNARMIC_SOURCE}/src/dynarmic/backend/x64/a32_emit_x64.cpp")
set(_vitox_dst "${CMAKE_BINARY_DIR}/vitox-patched/a32_emit_x64.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_vitox_src}")
file(READ "${_vitox_src}" _vitox_text)
string(REPLACE "\r\n" "\n" _vitox_text "${_vitox_text}") # checkout may be CRLF
set(_vitox_old [=[
    code.PreludeComplete();
    ClearFastDispatchTable();

    exception_handler.SetFastmemCallback([this](u64 rip_) {
        return FastmemCallback(rip_);
    });
}]=])
set(_vitox_new [=[
    // Vitox: must precede PreludeComplete(); on Windows this writes into the code cache.
    exception_handler.SetFastmemCallback([this](u64 rip_) {
        return FastmemCallback(rip_);
    });

    code.PreludeComplete();
    ClearFastDispatchTable();
}]=])
string(FIND "${_vitox_text}" "${_vitox_old}" _vitox_pos)
if (_vitox_pos EQUAL -1)
    message(FATAL_ERROR "Vitox shim: A32EmitX64 constructor no longer matches; re-check the W^X ordering fix")
endif()
string(REPLACE "${_vitox_old}" "${_vitox_new}" _vitox_text "${_vitox_text}")
file(WRITE "${_vitox_dst}.tmp" "${_vitox_text}")
file(COPY_FILE "${_vitox_dst}.tmp" "${_vitox_dst}" ONLY_IF_DIFFERENT)

get_target_property(_vitox_sources dynarmic SOURCES)
list(FIND _vitox_sources "backend/x64/a32_emit_x64.cpp" _vitox_idx)
if (_vitox_idx EQUAL -1)
    message(FATAL_ERROR "Vitox shim: backend/x64/a32_emit_x64.cpp not found in dynarmic SOURCES")
endif()
list(REMOVE_AT _vitox_sources ${_vitox_idx})
list(APPEND _vitox_sources "${_vitox_dst}")
set_property(TARGET dynarmic PROPERTY SOURCES ${_vitox_sources})
message(STATUS "Vitox shim: compiling patched ${_vitox_dst}")

# Candidate upstream/local integration patch; review before carrying it past M3.
# The A64 emitter (a64_emit_x64.cpp, A64EmitX64 ctor) has the same ordering; A64 is not built here.
set(VITOX_DYNARMIC_SHIMS "a32_emit_x64-fastmem-cb-before-prelude-complete")

# Record exactly what the probe links, for its logs (out\dynarmic\vitox-patched\vitox_dynarmic_build.h).
execute_process(COMMAND git -C "${VITOX_DYNARMIC_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE _vitox_commit OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git -C "${VITOX_DYNARMIC_SOURCE}" status --porcelain
    OUTPUT_VARIABLE _vitox_dirty OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if (_vitox_dirty)
    set(_vitox_tree "dirty")
else()
    set(_vitox_tree "clean")
endif()
if (DYNARMIC_ENABLE_NO_EXECUTE_SUPPORT)
    set(_vitox_nx 1)
else()
    set(_vitox_nx 0)
endif()
message(STATUS "Vitox Dynarmic: commit ${_vitox_commit} (${_vitox_tree}); shims: ${VITOX_DYNARMIC_SHIMS}")
file(WRITE "${CMAKE_BINARY_DIR}/vitox-patched/vitox_dynarmic_build.h.tmp"
"// Generated by Vitox-M2 dynarmic-build/CMakeLists.txt. Do not edit.
#pragma once
#define VITOX_DYNARMIC_COMMIT \"${_vitox_commit}\"
#define VITOX_DYNARMIC_TREE \"${_vitox_tree}\"
#define VITOX_DYNARMIC_SHIMS \"${VITOX_DYNARMIC_SHIMS}\"
#define VITOX_DYNARMIC_NO_EXECUTE_SUPPORT ${_vitox_nx}
")
file(COPY_FILE "${CMAKE_BINARY_DIR}/vitox-patched/vitox_dynarmic_build.h.tmp"
    "${CMAKE_BINARY_DIR}/vitox-patched/vitox_dynarmic_build.h" ONLY_IF_DIFFERENT)
